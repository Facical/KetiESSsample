1. 개요
--------------------------------------------------------------------------------

Vision Pro에서 보고 있는 ESS 3D 뷰를 웹 브라우저에서도 실시간으로 동일하게
볼 수 있도록 연동하는 방법을 설명합니다.

목표:
- Vision Pro 사용자가 보는 3D 뷰를 웹에서 실시간 확인
- 웹에서는 조작 없이 관찰만 가능 (View-Only)
- 실시간 ESS 데이터 동기화

[파일: server.js]

const express = require('express');
const { WebSocketServer } = require('ws');
const http = require('http');

const app = express();
const server = http.createServer(app);
const wss = new WebSocketServer({ server });

// 현재 뷰 상태 저장
let currentViewState = null;

// 연결된 클라이언트 관리
const clients = {
    visionPro: null,
    web: new Set()
};

wss.on('connection', (ws, req) => {
    const clientType = req.url.includes('visionpro') ? 'visionPro' : 'web';

    if (clientType === 'visionPro') {
        clients.visionPro = ws;
        console.log('Vision Pro 연결됨');
    } else {
        clients.web.add(ws);
        console.log(`웹 클라이언트 연결됨 (총 ${clients.web.size}개)`);

        // 새 웹 클라이언트에게 현재 상태 전송
        if (currentViewState) {
            ws.send(JSON.stringify(currentViewState));
        }
    }

    ws.on('message', (data) => {
        const message = JSON.parse(data);

        // Vision Pro에서 온 상태 업데이트
        if (clientType === 'visionPro' && message.type === 'viewState') {
            currentViewState = message;

            // 모든 웹 클라이언트에게 브로드캐스트
            clients.web.forEach(client => {
                if (client.readyState === 1) { // OPEN
                    client.send(JSON.stringify(message));
                }
            });
        }
    });

    ws.on('close', () => {
        if (clientType === 'visionPro') {
            clients.visionPro = null;
        } else {
            clients.web.delete(ws);
        }
    });
});

// ESS 데이터 API (기존 백엔드 연동용)
app.get('/api/ess/current', (req, res) => {
    // 실제로는 ESS 시스템에서 데이터 조회
    res.json({ /* ESS 데이터 */ });
});

server.listen(3000, () => {
    console.log('서버 시작: http://localhost:3000');
});

[파일: WebSyncManager.swift] - 새로 추가

import Foundation

class WebSyncManager: ObservableObject {
    private var webSocket: URLSessionWebSocketTask?
    private let serverURL = URL(string: "wss://your-server.com/ws?client=visionpro")!

    @Published var isConnected = false

    func connect() {
        let session = URLSession(configuration: .default)
        webSocket = session.webSocketTask(with: serverURL)
        webSocket?.resume()
        isConnected = true
        receiveMessage()
    }

    func disconnect() {
        webSocket?.cancel(with: .normalClosure, reason: nil)
        isConnected = false
    }

    // 뷰 상태 전송
    func sendViewState(_ state: ViewState) {
        guard let webSocket = webSocket else { return }

        let encoder = JSONEncoder()
        if let data = try? encoder.encode(state),
           let jsonString = String(data: data, encoding: .utf8) {
            let message = URLSessionWebSocketTask.Message.string(jsonString)
            webSocket.send(message) { error in
                if let error = error {
                    print("전송 오류: \(error)")
                }
            }
        }
    }

    private func receiveMessage() {
        webSocket?.receive { [weak self] result in
            switch result {
            case .success(let message):
                // 서버로부터 메시지 수신 처리 (필요시)
                self?.receiveMessage() // 계속 수신 대기
            case .failure(let error):
                print("수신 오류: \(error)")
            }
        }
    }
}

import Foundation
import simd

struct ViewState: Codable {
    let type: String = "viewState"
    let timestamp: Int64
    let camera: CameraState
    let rack: RackState
    let interaction: InteractionState
    let ui: UIState

    struct CameraState: Codable {
        let position: Vector3
        let rotation: Vector3

    struct RackState: Codable {
        let position: Vector3
        let rotation: Vector3
        let scale: Float
    }

    struct InteractionState: Codable {
        let selectedModuleIndex: Int?
        let pulledModules: [PulledModule]
    }

    struct PulledModule: Codable {

        let pullDistance: Float
    }

    struct UIState: Codable {
        let problemAnalysisEnabled: Bool
        let problemMetric: String
        let labelsVisible: Bool
        let selectedTab: String
    }

    struct Vector3: Codable {
        let x: Float
        let y: Float
        let z: Float

import { useEffect, useRef, useCallback } from 'react';
import { useViewStateStore } from '../stores/viewStateStore';

export function useWebSocket(serverUrl: string) {
    const ws = useRef<WebSocket | null>(null);
    const { setViewState, setESSData, setConnected } = useViewStateStore();

    useEffect(() => {
        ws.current = new WebSocket(serverUrl);

        ws.current.onopen = () => {
            console.log('서버 연결됨');
            setConnected(true);
        };

        ws.current.onmessage = (event) => {
            const message = JSON.parse(event.data);

            switch (message.type) {
                case 'viewState':
                    setViewState(message);
                    break;
                case 'essData':
                    setESSData(message);
                    break;
            }
        };

        ws.current.onclose = () => {
            setConnected(false);
            // 재연결 로직
            setTimeout(() => {
                // reconnect
            }, 3000);
        };

        return () => {
            ws.current?.close();
        };
    }, [serverUrl]);

    return { isConnected: ws.current?.readyState === WebSocket.OPEN };
}

import { create } from 'zustand';

interface ViewState {
    camera: {
        position: { x: number; y: number; z: number };
        rotation: { x: number; y: number; z: number };
    };
    rack: {
        position: { x: number; y: number; z: number };

        selectedModuleIndex: number | null;

        problemAnalysisEnabled: boolean;
        labelsVisible: boolean;
    };
}

interface Store {
    viewState: ViewState | null;
    essData: any;
    isConnected: boolean;
    setViewState: (state: ViewState) => void;
    setESSData: (data: any) => void;
    setConnected: (connected: boolean) => void;
}

export const useViewStateStore = create<Store>((set) => ({
    viewState: null,
    essData: null,
    isConnected: false,
    setViewState: (state) => set({ viewState: state }),
    setESSData: (data) => set({ essData: data }),
    setConnected: (connected) => set({ isConnected: connected }),
}));

import React, { useRef, useEffect } from 'react';
import { Canvas, useFrame, useThree } from '@react-three/fiber';
import { OrbitControls, useGLTF } from '@react-three/drei';
import { useViewStateStore } from '../stores/viewStateStore';

// 카메라 동기화 컴포넌트
function SyncedCamera() {
    const { camera } = useThree();
    const viewState = useViewStateStore((state) => state.viewState);

    useFrame(() => {
        if (viewState?.camera) {
            camera.position.set(
                viewState.camera.position.x,
                viewState.camera.position.y,
                viewState.camera.position.z
            );
            camera.rotation.set(
                viewState.camera.rotation.x * Math.PI / 180,
                viewState.camera.rotation.y * Math.PI / 180,
                viewState.camera.rotation.z * Math.PI / 180
            );
        }
    });

    return null;
}

// ESS 랙 모델 컴포넌트
function RackModel() {
    const { scene } = useGLTF('/models/rack.glb');
    const rackRef = useRef();
    const viewState = useViewStateStore((state) => state.viewState);

    useFrame(() => {
        if (rackRef.current && viewState?.rack) {
            rackRef.current.position.set(
                viewState.rack.position.x,
                viewState.rack.position.y,
                viewState.rack.position.z
            );
            rackRef.current.scale.setScalar(viewState.rack.scale);
        }
    });

    return <primitive ref={rackRef} object={scene} />;
}

// 모듈 상태 시각화
function ModuleHighlights() {
    const viewState = useViewStateStore((state) => state.viewState);
    const essData = useViewStateStore((state) => state.essData);

    if (!viewState?.interaction?.pulledModules) return null;

    return (
        <>
            {viewState.interaction.pulledModules.map((module) => {
                const moduleData = essData?.modules?.[module.index];
                const status = moduleData?.status || 'normal';
                const color = status === 'critical' ? '#ff0000'
                            : status === 'warning' ? '#ffa500'
                            : '#00ff00';

                return (
                    <mesh
                        key={module.index}
                        position={[
                            /* 모듈 위치 계산 */
                            0,
                            module.index * 0.15,
                            module.pullDistance
                        ]}
                    >
                        <boxGeometry args={[0.3, 0.1, 0.4]} />
                        <meshStandardMaterial color={color} />
                    </mesh>
                );
            })}
        </>
    );
}

4. 구현 단계별 로드맵
================================================================================

[Phase 1: 기반 구축]
- 백엔드 서버 세팅 (Node.js + WebSocket)
- Vision Pro 앱에 WebSyncManager 추가
- 기본 연결 테스트

[Phase 2: 상태 동기화]
- ViewState 데이터 구조 정의
- Vision Pro → 서버 상태 전송 구현
- 서버 → 웹 브로드캐스트 구현

[Phase 3: 웹 뷰어 개발]
- React + Three.js 프로젝트 세팅
- 3D 모델 변환 및 로드
- 상태 기반 렌더링 구현

[Phase 4: ESS 데이터 통합]
- 실시간 ESS 데이터 스트리밍
- 데이터 패널 UI 구현
- Problem Analysis 시각화

[Phase 5: 최적화 및 배포]
- 네트워크 지연 최적화
- 보안 (인증, HTTPS)
- 클라우드 배포


================================================================================
5. 고려사항 및 제약
================================================================================

[성능]
- WebSocket 메시지 주기: 30-60fps 권장 (약 16-33ms 간격)
- 대역폭: 상태 데이터만 전송 시 약 1-5KB/초
- 지연: 같은 네트워크 내 <50ms 목표

[보안]
- WSS (WebSocket Secure) 사용 필수
- 인증 토큰 기반 연결 관리
- 민감한 ESS 데이터 암호화 고려

[확장성]
- 여러 웹 클라이언트 동시 연결 지원
- 여러 Vision Pro 기기 지원 (세션 관리 필요)
- 녹화/재생 기능 추가 가능

[제약]
- 웹에서 visionOS와 완전히 동일한 렌더링은 불가능
- 손 제스처 등 AR 고유 기능은 웹에서 시각화만 가능
- 네트워크 불안정 시 동기화 지연 발생
